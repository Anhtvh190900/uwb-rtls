# Headless profile for the vehicle embedded platform (Jetson Orin, aarch64).
#
# The Orin runs no GUI: RTLS Studio, the Programmer and the simulation tools all
# stay on a desktop machine. What is left is the protocol layer plus a serial
# link, which is everything a data-collection or debug script needs.
#
# This list was derived by reading the actual imports of software/common/ and
# software/vv_testings/, not by trimming requirements.txt by hand.
#
# Install with:  python3 install.py --profile orin

protobuf>=6.31.1     # google.protobuf - common/protocol_pb2.py (gencode 6.31.1)
pyserial>=3.5        # serial - common/transport.py, vv_testings/*

#                        silently resolves to 6.8.0.2 here and cannot move.
